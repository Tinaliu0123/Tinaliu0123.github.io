<script>
  (function () {
    var imgs = Array.prototype.slice.call(document.querySelectorAll(".misc-collage img[data-zoom-src]"));
    if (!imgs.length) return;
    var reduce = window.matchMedia("(prefers-reduced-motion: reduce)").matches;
    var SLIDE_MS = reduce ? 0 : 560, WHEEL_LOCK_MS = 650, WHEEL_GAP_MS = 200, WHEEL_MIN = 28, SWIPE_MIN = 40;
    var pad = function (n) { return (n < 10 ? "0" : "") + n; };
    var mv = document.createElement("div");
    mv.className = "mv"; mv.setAttribute("role", "dialog"); mv.setAttribute("aria-modal", "true"); mv.setAttribute("aria-label", "Photo viewer");
    mv.innerHTML = '<div class="mv__count" aria-live="polite"></div><button type="button" class="mv__close" aria-label="Close">&times;</button>';
    var strip = document.createElement("div");
    strip.className = "mv__strip"; strip.setAttribute("role", "group"); strip.setAttribute("aria-label", "Photos");
    var bars = imgs.map(function (g, i) {
      var b = document.createElement("button");
      b.type = "button"; b.className = "mv__bar"; b.setAttribute("aria-label", "Photo " + (i + 1));
      b.appendChild(document.createElement("i"));
      b.addEventListener("click", function (e) { e.stopPropagation(); jump(i); });
      strip.appendChild(b); return b;
    });
    mv.appendChild(strip);
    document.body.appendChild(mv);
    var countEl = mv.querySelector(".mv__count"), closeBtn = mv.querySelector(".mv__close");
    var cur = -1, shown = null, savedY = 0, busyUntil = 0, colours = {};
    function avg(img) {
      try {
        var c = document.createElement("canvas"); c.width = c.height = 16;
        var x = c.getContext("2d"); x.drawImage(img, 0, 0, 16, 16);
        var d = x.getImageData(0, 0, 16, 16).data, r = 0, g = 0, b = 0;
        for (var i = 0; i < d.length; i += 4) { r += d[i]; g += d[i + 1]; b += d[i + 2]; }
        var n = d.length / 4; return [r / n / 255, g / n / 255, b / n / 255];
      } catch (e) { return null; }
    }
    function tone(rgb) {
      var r = rgb[0], g = rgb[1], b = rgb[2], mx = Math.max(r, g, b), mn = Math.min(r, g, b), l = (mx + mn) / 2, h = 0, s = 0;
      if (mx !== mn) {
        var dd = mx - mn; s = l > 0.5 ? dd / (2 - mx - mn) : dd / (mx + mn);
        h = (mx === r ? (g - b) / dd + (g < b ? 6 : 0) : mx === g ? (b - r) / dd + 2 : (r - g) / dd + 4) * 60;
      }
      var dark = l < 0.36, sat = Math.min(0.52, s * 1.8 + 0.06), hh = h.toFixed(0);
      var pct = function (v) { return (v * 100).toFixed(0) + "%"; };
      return {
        dark: dark, ink: dark ? "#f3f0ea" : "#1d1d1f",
        bg: "hsl(" + hh + " " + pct(sat) + " " + (dark ? 19 : 82) + "%)",
        raw: "rgb(" + Math.round(r * 255) + ", " + Math.round(g * 255) + ", " + Math.round(b * 255) + ")",
        tint: ["hsl(" + hh + " " + pct(Math.min(0.4, s * 1.4 + 0.06)) + " 93%)", "hsl(" + hh + " " + pct(Math.min(0.3, s + 0.05)) + " 11%)"]
      };
    }
    function store(i, el) {
      if (colours[i] === undefined) {
        var a = avg(el); colours[i] = a ? tone(a) : null;
        if (colours[i]) bars[i].style.setProperty("--c", colours[i].raw);
      }
      return colours[i];
    }
    function colourFor(i, then) {
      if (colours[i] !== undefined) return then(colours[i]);
      var g = imgs[i];
      if (g.complete && g.naturalWidth) return then(store(i, g));
      var probe = new Image(); probe.onload = function () { then(store(i, probe)); }; probe.onerror = function () { then(null); }; probe.src = g.currentSrc || g.src;
    }
    imgs.forEach(function (g, i) {
      if (g.complete && g.naturalWidth) store(i, g); else g.addEventListener("load", function () { store(i, g); }, { once: true });
    });
    var filling = false;
    function fillMissing() {
      if (filling) return; filling = true;
      var todo = imgs.map(function (g, i) { return i; }).filter(function (i) { return colours[i] === undefined; });
      var next = function () { var i = todo.shift(); if (i === undefined) { filling = false; return; } colourFor(i, next); };
      next(); next();
    }
    function applyColour(i) {
      colourFor(i, function (t) {
        if (i !== cur || !t) return;
        mv.style.setProperty("--mv-bg", t.bg); mv.style.setProperty("--mv-ink", t.ink);
        mv.style.setProperty("--mv-edge", t.dark ? "rgba(255, 255, 255, 0.3)" : "rgba(0, 0, 0, 0.1)");
        mv.style.setProperty("--mv-dim", t.dark ? "0.75" : "0.55");
      });
    }
    function box(img) {
      var vw = window.innerWidth, vh = window.innerHeight, phone = vw < 600;
      var ar = (img.naturalWidth || 3) / (img.naturalHeight || 2);
      var maxW = phone ? vw - 32 : vw * 0.7, maxH = phone ? vh * 0.7 : vh * 0.72;
      var w = maxW, h = w / ar; if (h > maxH) { h = maxH; w = h * ar; }
      return { w: Math.round(w), h: Math.round(h) };
    }
    function makeImg(i) {
      var g = imgs[i], el = document.createElement("img");
      el.className = "mv__img"; el.alt = ""; el.draggable = false;
      var b = box(g); el.style.width = b.w + "px"; el.style.height = b.h + "px";
      el.src = g.currentSrc || g.src;
      var hi = new Image(); hi.onload = function () { el.src = hi.src; }; hi.src = g.getAttribute("data-zoom-src");
      return el;
    }
    function preload(i) {
      [i - 1, i + 1].forEach(function (k) { var g = imgs[(k + imgs.length) % imgs.length]; new Image().src = g.getAttribute("data-zoom-src"); });
    }
    function setCount(i) {
      countEl.textContent = pad(i + 1) + " / " + pad(imgs.length);
      bars.forEach(function (b, k) { b.classList.toggle("is-cur", k === i); if (k === i) b.setAttribute("aria-current", "true"); else b.removeAttribute("aria-current"); });
    }
    function shift(axis, k) { return axis === "y" ? "translate(-50%, calc(-50% + " + (k * 12) + "vh))" : "translate(calc(-50% + " + (k * 12) + "vw), -50%)"; }
    function step(delta, axis) {
      if (cur < 0 || Date.now() < busyUntil) return;
      var next = (cur + delta + imgs.length) % imgs.length, off = delta > 0 ? 1 : -1;
      busyUntil = Date.now() + SLIDE_MS;
      var incoming = makeImg(next), outgoing = shown;
      incoming.style.opacity = "0"; incoming.style.transform = shift(axis, off);
      mv.insertBefore(incoming, countEl);
      cur = next; shown = incoming; setCount(cur); applyColour(cur); preload(cur);
      requestAnimationFrame(function () { requestAnimationFrame(function () {
        incoming.style.opacity = "1"; incoming.style.transform = "translate(-50%, -50%)";
        if (outgoing) { outgoing.style.opacity = "0"; outgoing.style.transform = shift(axis, -off); }
      }); });
      setTimeout(function () { if (outgoing && outgoing.parentNode) outgoing.parentNode.removeChild(outgoing); }, SLIDE_MS + 60);
    }
    function jump(j) { if (cur < 0 || j === cur) return; step(j - cur, window.innerWidth < 600 ? "x" : "y"); }
    function open(i) {
      cur = i; savedY = window.pageYOffset;
      ["--mv-bg", "--mv-ink", "--mv-edge", "--mv-dim"].forEach(function (p) { mv.style.removeProperty(p); });
      Array.prototype.forEach.call(mv.querySelectorAll(".mv__img"), function (el) { el.parentNode.removeChild(el); });
      shown = makeImg(i); shown.style.transform = reduce ? "translate(-50%, -50%)" : "translate(-50%, -50%) scale(0.96)";
      mv.insertBefore(shown, countEl);
      setCount(i); applyColour(i); preload(i);
      document.documentElement.classList.add("mv-lock");
      mv.classList.add("is-open");
      requestAnimationFrame(function () { requestAnimationFrame(function () { if (shown) shown.style.transform = "translate(-50%, -50%)"; }); });
      closeBtn.focus({ preventScroll: true });
      setTimeout(fillMissing, 400);
    }
    function close() {
      if (cur < 0) return;
      cur = -1; mv.classList.remove("is-open");
      document.documentElement.classList.remove("mv-lock");
      window.scrollTo(0, savedY);
    }
    imgs.forEach(function (g, i) {
      g.addEventListener("click", function () { open(i); });
      g.setAttribute("tabindex", "0"); g.setAttribute("role", "button"); g.setAttribute("aria-label", "Open photo " + (i + 1));
      g.addEventListener("keydown", function (e) { if (e.key === "Enter" || e.key === " ") { e.preventDefault(); open(i); } });
    });
    closeBtn.addEventListener("click", function (e) { e.stopPropagation(); close(); });
    var swiped = false;
    mv.addEventListener("click", function (e) {
      if (swiped) { swiped = false; return; }
      if (e.target === mv) close();
    });
    document.addEventListener("keydown", function (e) {
      if (cur < 0) return;
      var k = e.key;
      if (k === "Escape") { e.preventDefault(); close(); }
      else if (k === "ArrowRight") { e.preventDefault(); step(1, "x"); }
      else if (k === "ArrowLeft") { e.preventDefault(); step(-1, "x"); }
      else if (k === "ArrowDown" || k === "PageDown") { e.preventDefault(); step(1, "y"); }
      else if (k === "ArrowUp" || k === "PageUp") { e.preventDefault(); step(-1, "y"); }
    });
    var acc = 0, lastEvt = 0, lastStep = 0, inGesture = false;
    mv.addEventListener("wheel", function (e) {
      e.preventDefault();
      var now = Date.now(), gap = now - lastEvt; lastEvt = now;
      if (inGesture && gap < WHEEL_GAP_MS) return;
      inGesture = false;
      if (gap >= WHEEL_GAP_MS) acc = 0;
      if (now - lastStep < WHEEL_LOCK_MS) { acc = 0; inGesture = true; return; }
      var unit = e.deltaMode === 1 ? 16 : 1, horiz = Math.abs(e.deltaX) > Math.abs(e.deltaY);
      acc += (horiz ? e.deltaX : e.deltaY) * unit;
      if (Math.abs(acc) < WHEEL_MIN) return;
      step(acc > 0 ? 1 : -1, horiz ? "x" : "y");
      acc = 0; lastStep = now; inGesture = true;
    }, { passive: false });
    var tx = 0, ty = 0;
    mv.addEventListener("touchstart", function (e) { var t = e.touches[0]; tx = t.clientX; ty = t.clientY; }, { passive: true });
    mv.addEventListener("touchend", function (e) {
      var t = e.changedTouches[0], dx = t.clientX - tx, dy = t.clientY - ty;
      if (Math.max(Math.abs(dx), Math.abs(dy)) < SWIPE_MIN) return;
      swiped = true; setTimeout(function () { swiped = false; }, 400);
      if (Math.abs(dx) > Math.abs(dy)) step(dx < 0 ? 1 : -1, "x"); else step(dy < 0 ? 1 : -1, "y");
    }, { passive: true });
    if (window.matchMedia("(hover: hover) and (pointer: fine)").matches) {
      var hoverI = -1, collage = document.querySelector(".misc-collage");
      imgs.forEach(function (g, i) {
        g.addEventListener("mouseenter", function () {
          hoverI = i; document.documentElement.classList.add("misc-tinting");
          colourFor(i, function (t) {
            if (!t || hoverI !== i) return;
            document.body.style.setProperty("--global-bg-color", t.tint[document.documentElement.getAttribute("data-theme") === "dark" ? 1 : 0]);
          });
        });
      });
      if (collage) collage.addEventListener("mouseleave", function () { hoverI = -1; document.body.style.removeProperty("--global-bg-color"); });
    }
    window.addEventListener("resize", function () {
      if (cur < 0 || !shown) return; var b = box(imgs[cur]); shown.style.width = b.w + "px"; shown.style.height = b.h + "px";
    });
  })();
